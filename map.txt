function chaveTemporada(auth) {
    return "season_" + temporadaAtual + "_" + auth;

function getSeasonStats(auth) {
    if (!temporadaStats[auth]) {
        temporadaStats[auth] = {
            wins: 0,
            losses: 0,
            draws: 0,
            goals: 0,
            assists: 0,
            games: 0,
            mvp: 0,
            points: 0
        };
    }
    return temporadaStats[auth];
}

function registrarStatTemporada(player, tipo, quantidade) {
    if (!player) return;
    var auth = getAuth(player);
    var s = getSeasonStats(auth);
    quantidade = Number(quantidade) || 0;

    if (tipo === "win") s.wins += quantidade;
    if (tipo === "loss") s.losses += quantidade;
    if (tipo === "draw") s.draws += quantidade;
    if (tipo === "goal") s.goals += quantidade;
    if (tipo === "assist") s.assists += quantidade;
    if (tipo === "game") s.games += quantidade;
    if (tipo === "mvp") s.mvp += quantidade;

    // Pontos da temporada:
    // vitória = 3, empate = 1, gol = 1, assistência = 1, MVP = 3.
    if (tipo === "win") s.points += 3 * quantidade;
    if (tipo === "draw") s.points += 1 * quantidade;
    if (tipo === "goal") s.points += 1 * quantidade;
    if (tipo === "assist") s.points += 1 * quantidade;
    if (tipo === "mvp") s.points += 3 * quantidade;
}

function rankingTemporadaAtual() {
    return Object.keys(temporadaStats).map(function(auth) {
        var s = temporadaStats[auth];
        var profile = getProfile(auth);
        return {
            auth: auth,
            stats: s,
            name: profile.name || "Jogador",
            points: s.points || 0
        };
    }).sort(function(a, b) {
        if (b.points !== a.points) return b.points - a.points;
        if (b.stats.wins !== a.stats.wins) return b.stats.wins - a.stats.wins;
        if (b.stats.goals !== a.stats.goals) return b.stats.goals - a.stats.goals;
        return b.stats.assists - a.stats.assists;
    });
}

function resetarContaParaNovaTemporada(auth) {
    var profile = getProfile(auth);

    // Tudo que pertence à temporada é zerado/removido.
    profile.coins = 0;
    profile.purchases = [];
    profile.title = "";
    profile.booster = false;
    profile.streak = 0;
    profile.bestStreak = 0;

    // Campos estatísticos persistentes conhecidos do sistema.
    profile.wins = 0;
    profile.losses = 0;
    profile.draws = 0;
    profile.games = 0;
    profile.goals = 0;
    profile.assists = 0;
    profile.mvp = 0;
    profile.rankPoints = 0;

    saveProfile(auth, profile);
}

function fecharTemporada() {
    var ranking = rankingTemporadaAtual();
    var campeao = ranking[0] || null;

    historicoTemporadas.push({
        temporada: temporadaAtual,
        inicio: temporadaInicio,
        fim: Date.now(),
        campeao: campeao ? campeao.name : "Sem campeão",
        campeaoAuth: campeao ? campeao.auth : null,
        ranking: ranking.map(function(x) {
            return {
                name: x.name,
                auth: x.auth,
                wins: x.stats.wins,
                losses: x.stats.losses,
                draws: x.stats.draws,
                goals: x.stats.goals,
                assists: x.stats.assists,
                games: x.stats.games,
                mvp: x.stats.mvp,
                points: x.points
            };
        })
    });

    if (campeao) {
        room.sendAnnouncement("🏆 TEMPORADA " + temporadaAtual + " ENCERRADA! Campeão: " + campeao.name, null, Cor.Ouro, "bold", 2);
    }

    // Zera contas de TODOS que participaram da temporada.
    Object.keys(temporadaStats).forEach(function(auth) {
        resetarContaParaNovaTemporada(auth);
    });

    temporadaStats = {};
    temporadaAtual++;
    temporadaInicio = Date.now();

    room.sendAnnouncement("🔥 TEMPORADA " + temporadaAtual + " COMEÇOU! Boa sorte!", null, Cor.Verde, "bold", 2);
}

function verificarFimTemporada() {
    if (Date.now() - temporadaInicio >= TEMPORADA_DURACAO_DIAS * 24 * 60 * 60 * 1000) {
        fecharTemporada();
        return true;
    }
    return false;
}

function temporadaAnterior() {
    return historicoTemporadas.length

        : null;
}

function authEstaNoCargo(player, config) {
    if (!player || !config || !Array.isArray(config.auths)) return false;
    var auth = (typeof getAuth === "function" ? getAuth(player) : null) || player.auth;
    return !!auth && config.auths.indexOf(auth) !== -1;
}

function hexParaCor(hex, fallback) {
    // Aceita: "#FF0000", "FF0000", "0xFF0000" ou 0xFF0000.
    if (typeof hex === "number" && isFinite(hex)) {
        return hex & 0xFFFFFF;
    }
    if (typeof hex !== "string") return fallback;

    var valor = hex.trim()
        .replace(/^#/, "")
        .replace(/^0x/i, "");

    if (!/^[0-9a-fA-F]{6}$/.test(valor)) return fallback;
    return parseInt(valor, 16);
}

function configCargo(player) {
    if (!player) return null;

    // O AUTH original do HaxBall é a fonte principal do cargo.
    // O fallback pelo extendedP existe apenas para casos em que player.auth não estiver disponível.
    var authCargo = player.auth || ((typeof getAuth === "function" ? getAuth(player) : null));
    if (!authCargo) return null;

    function achar(config, nome, fallback) {
        if (!config || !Array.isArray(config.auths)) return null;
        var i = config.auths.indexOf(authCargo);
        if (i === -1) return null;

        var p = (Array.isArray(config.personalizacao) && config.personalizacao[i])
            ? config.personalizacao[i]
            : fallback;

        return {
            cargo: nome,
            index: i,
            cor: p.cor,
            emoji: p.emoji
        };
    }

    var cargoConfigurado = achar(SUPERADMIN_CONFIG, "SuperAdmin", {cor:"#000000", emoji:"👑"}) ||
           achar(ADMIN_CONFIG, "Admin", {cor:"#44daf3", emoji:"🛡️"}) ||
           achar(BOOSTER_CONFIG, "Booster", {cor:"#b84dff", emoji:"💎"});

    if (cargoConfigurado) return cargoConfigurado;

    // Booster comprado na loja fica salvo no perfil e continua válido após reconectar.
    var perfilCargo = getProfile(authCargo);
    if (perfilCargo && perfilCargo.booster === true) {
        return { cargo: "Booster", index: -1, cor: "#b84dff", emoji: "💎" };
    }

    return null;
}

function discordPostJSON(webhookURL, payload) {
    try {
        if (!webhookURL) return;
        var xhr = new XMLHttpRequest();
        xhr.open("POST", webhookURL, true);
        xhr.setRequestHeader("Content-Type", "application/json");
        xhr.send(JSON.stringify(payload));
    } catch (e) {
        console.log("[WEBHOOK] Erro ao enviar JSON:", e);
    }
}

function getProfile(auth) {
    var key = "amk_profile_" + auth;
    var raw = localStorage.getItem(key);
    var profile = raw ? JSON.parse(raw) : null;
    if (!profile) profile = { coins: 100, streak: 0, bestStreak: 0, purchases: [], title: "", booster: false, coinsDay: dataCoinsAtual(), coinsToday: 0, lastCoinGame: "" };
    if (!Array.isArray(profile.purchases)) profile.purchases = [];
    if (typeof profile.title !== "string") profile.title = "";
    if (typeof profile.booster !== "boolean") profile.booster = false;
    if (typeof profile.coins !== "number") profile.coins = 100;
    if (typeof profile.streak !== "number") profile.streak = 0;
    if (typeof profile.bestStreak !== "number") profile.bestStreak = 0;
    if (typeof profile.coinsDay !== "string") profile.coinsDay = dataCoinsAtual();
    if (typeof profile.coinsToday !== "number") profile.coinsToday = 0;
    if (typeof profile.lastCoinGame !== "string") profile.lastCoinGame = "";
    if (profile.coinsDay !== dataCoinsAtual()) {
        profile.coinsDay = dataCoinsAtual();
        profile.coinsToday = 0;
    }
    return profile;
}

function saveProfile(auth, profile) {
    localStorage.setItem("amk_profile_" + auth, JSON.stringify(profile));
}

function addCoins(player, amount) {
    if (!player) return;
    var profile = getProfile(getAuth(player));
    profile.coins += Math.max(0, Number(amount) || 0);
    saveProfile(getAuth(player), profile);
}

function encontrarJogadorPorNome(nome) {
    if (nome === undefined || nome === null) return null;

    var alvo = String(nome).trim();
    if (!alvo) return null;

    var lista = room.getPlayerList().filter(function(p) { return p.id !== 0; });

    // Aceita !addcoins #1 75000 ou !addcoins 1 75000
    var idTexto = alvo.replace(/^#/, "");
    if (/^\d+$/.test(idTexto)) {
        var id = Number(idTexto);
        var porId = lista.find(function(p) { return p.id === id; });
        if (porId) return porId;
    }

    var alvoLower = alvo.toLowerCase();

    // Nome exato primeiro.
    var exato = lista.find(function(p) {
        return String(p.name).toLowerCase() === alvoLower;
    });
    if (exato) return exato;

    // Depois aceita começo do nick.
    return lista.find(function(p) {
        return String(p.name).toLowerCase().indexOf(alvoLower) === 0;
    }) || null;
}

// RECOMPENSAS DE COINS — PAGAS SOMENTE AO FINAL DA PARTIDA
var coinsAwardedForGame = false;

function dataCoinsAtual() {
    var d = new Date();
    return d.getFullYear() + "-" + (d.getMonth() + 1) + "-" + d.getDate();
}

function prepararProfileCoins(profile) {
    var hoje = dataCoinsAtual();
    if (profile.coinsDay !== hoje) {
        profile.coinsDay = hoje;
        profile.coinsToday = 0;
    }
    if (typeof profile.coinsToday !== "number") profile.coinsToday = 0;
    if (typeof profile.lastCoinGame !== "string") profile.lastCoinGame = "";
    return profile;
}

function addCoinsSeguro(player, amount, gameId) {
    if (!player || amount <= 0) return 0;

    var auth = getAuth(player);
    var profile = prepararProfileCoins(getProfile(auth));

    // Uma mesma partida nunca pode pagar duas vezes para o mesmo jogador.
    if (gameId && profile.lastCoinGame === gameId) return 0;

    var disponivel = Math.max(0, COINS_DAILY_CAP - profile.coinsToday);
    var creditado = Math.min(Number(amount) || 0, disponivel);

    if (creditado > 0) {
        profile.coins += creditado;
        profile.coinsToday += creditado;
        if (gameId) profile.lastCoinGame = gameId;
        saveProfile(auth, profile);
    }
    return creditado;
}

function premiarCoinsDaPartida(mvp) {
    if (coinsAwardedForGame || !game || !game.scores) return;
    if (!allReds.length || !allBlues.length) return;

    // Só remunera partidas 4x4 válidas.
    if (allReds.length !== maxTeamSize || allBlues.length !== maxTeamSize) return;

    var duracao = Number(game.scores.time) || 0;
    var terminouPorGols = game.scores.red >= game.scores.scoreLimit || game.scores.blue >= game.scores.scoreLimit;
    if (duracao < COINS_MIN_GAME_SECONDS && !terminouPorGols) {
        sendGeneralLog("COINS BLOQUEADOS", "Partida muito curta (" + duracao + "s). Proteção anti-farm ativada.");
        coinsAwardedForGame = true;
        return;
    }

    var jogadores = allReds.concat(allBlues);
    var ids = {};
    jogadores.forEach(function(p) { if (p) ids[p.id] = true; });

    // ID único desta partida.
    var gameId = String(game.date || Date.now()) + ":" + String(game.scores.time || 0);

    // Vitória.
    if (lastWinner === Team.RED) {
        allReds.forEach(function(p) { addCoinsSeguro(p, COINS_VITORIA, gameId); });
    } else if (lastWinner === Team.BLUE) {
        allBlues.forEach(function(p) { addCoinsSeguro(p, COINS_VITORIA, gameId); });
    }

    // Gols e assistências, limitados aos jogadores da partida.
    var golsPagos = {};
    var assistsPagas = {};
    (game.goals || []).forEach(function(g) {
        if (g.striker && ids[g.striker.id]) {
            golsPagos[g.striker.id] = (golsPagos[g.striker.id] || 0) + 1;
        }
        if (g.assist && ids[g.assist.id]) {
            assistsPagas[g.assist.id] = (assistsPagas[g.assist.id] || 0) + 1;
        }
    });

    Object.keys(golsPagos).forEach(function(id) {
        var p = jogadores.find(function(x) { return x.id == id; });
        if (p) {
            // Cada jogador recebe no máximo 3 recompensas de gol por partida.
            var qtd = Math.min(golsPagos[id], 3);
            for (var i = 0; i < qtd; i++) addCoinsSeguro(p, COINS_GOL, gameId + ":G:" + i);
        }
    });

    Object.keys(assistsPagas).forEach(function(id) {
        var p = jogadores.find(function(x) { return x.id == id; });
        if (p) {
            var qtd = Math.min(assistsPagas[id], 3);
            for (var i = 0; i < qtd; i++) addCoinsSeguro(p, COINS_ASSISTENCIA, gameId + ":A:" + i);
        }
    });

    // Bônus de MVP, também protegido pelo limite diário.
    if (mvp && mvp.player) {
        addCoinsSeguro(mvp.player, COINS_MVP, gameId + ":MVP");
    }

    coinsAwardedForGame = true;
}

function atualizarSequenciasJogadores(winner) {
    var todos = [];
    if (allReds) todos = todos.concat(allReds);
    if (allBlues) todos = todos.concat(allBlues);
    var seen = {};
    todos.forEach(function(p) {
        if (!p || seen[p.id]) return;
        seen[p.id] = true;
        var profile = getProfile(getAuth(p));
        if (winner === Team.RED && p.team === Team.RED || winner === Team.BLUE && p.team === Team.BLUE) {
            profile.streak++;
            if (profile.streak > profile.bestStreak) profile.bestStreak = profile.streak;
        } else if (winner === Team.RED || winner === Team.BLUE) {
            profile.streak = 0;
        } else {
            profile.streak = 0;
        }
        saveProfile(getAuth(p), profile);
    });
}

function formatGoalList() {
    if (!game || !game.goals || !game.goals.length) return "Nenhum gol registrado.";
    return game.goals.map(function(g, i) {
        var striker = g.striker ? g.striker.name : "Gol contra";
        var assist = g.assist ? " | 👟 " + g.assist.name : "";
        var min = g.time != null ? Math.floor(g.time / 60) : 0;
        var sec = g.time != null ? Math.floor(g.time % 60) : 0;
        return (i + 1) + ". " + striker + assist + " — " + ("0" + min).slice(-2) + ":" + ("0" + sec).slice(-2);
    }).join("\n");
}


function calcularMVPPartida() {
    if (!game || !game.goals) return null;

    var jogadores = [];
    var porId = {};

    allReds.concat(allBlues).forEach(function(p) {
        if (!p || porId[p.id]) return;
        porId[p.id] = {
            player: p,
            goals: 0,
            assists: 0,
            team: p.team,
            cleanSheet: false,
            score: 0
        };
        jogadores.push(porId[p.id]);
    });

    game.goals.forEach(function(g) {
        if (g.striker && porId[g.striker.id]) porId[g.striker.id].goals++;
        if (g.assist && porId[g.assist.id]) porId[g.assist.id].assists++;
    });

    jogadores.forEach(function(s) {
        s.score = s.goals * 3 + s.assists * 2;
        if ((lastWinner === Team.RED && s.team === Team.RED) ||
            (lastWinner === Team.BLUE && s.team === Team.BLUE)) {
            s.score += 2;
        }
        if (s.team === Team.RED && game.scores.blue === 0) s.score += 2;
        if (s.team === Team.BLUE && game.scores.red === 0) s.score += 2;
    });

    jogadores.sort(function(a, b) {
        if (b.score !== a.score) return b.score - a.score;
        if (b.goals !== a.goals) return b.goals - a.goals;
        return b.assists - a.assists;
    });

    return jogadores.length && jogadores[0].score > 0 ? jogadores[0] : null;
}

function formatPlayerMatchStats() {
    var jogadores = [];
    var seen = {};

    allReds.concat(allBlues).forEach(function(p) {
        if (!p || seen[p.id]) return;
        seen[p.id] = true;
        var goals = 0, assists = 0;

        (game.goals || []).forEach(function(g) {
            if (g.striker && g.striker.id === p.id) goals++;
            if (g.assist && g.assist.id === p.id) assists++;
        });

        var gk = (GKList[0] && GKList[0].id === p.id) || (GKList[1] && GKList[1].id === p.id);
        jogadores.push({
            player: p,
            goals: goals,
            assists: assists,
            gk: gk
        });
    });

    jogadores.sort(function(a,b) {
        return (b.goals * 3 + b.assists * 2) - (a.goals * 3 + a.assists * 2);
    });

    return jogadores.map(function(s) {
        return (s.player.team === Team.RED ? "🔴 " : "🔵 ") +
            s.player.name + " — ⚽ " + s.goals + " | 👟 " + s.assists + (s.gk ? " | 🧤 GK" : "");
    }).join("\n");
}

    var redPoss = isFinite(Rposs) ? (Rposs * 100).toFixed(1) : "0.0";

    var vencedor = lastWinner === Team.RED ? "🔴 Red" : lastWinner === Team.BLUE ? "🔵 Blue" : "🤝 Empate";

        "**Placar:** 🔴 " + s.red + " x " + s.blue + " 🔵\n" +
        "**Vencedor:** " + vencedor + "\n" +
        "**Posse:** 🔴 " + redPoss + "% | 🔵 " + bluePoss + "%\n" +
        "**Duração:** " + Math.floor((s.time || 0) / 60) + ":" + ("0" + Math.floor((s.time || 0) % 60)).slice(-2) + "\n" +
        mvpTexto + "\n\n" +
        "⚽ **GOLS E ASSISTÊNCIAS**\n" + formatGoalList() + "\n\n" +
        "📊 **ESTATÍSTICAS DOS JOGADORES**\n" + (formatPlayerMatchStats() || "Sem estatísticas.") + "\n\n" +

        return false;

        room.sendAnnouncement("❌ Você já possui o Booster!", player.id, 0xFF6B6B, "bold");
        return false;
    }

    profile.coins -= item.preco;
    if (item.bonus) {
        profile.coins += item.bonus;
    } else {
        profile.purchases.push(itemKey);
        if (item.title) profile.title = item.title;
        if (item.avatar) room.setPlayerAvatar(player.id, item.avatar);
        if (item.role === "booster") {
            profile.booster = true;
            if (boosterPlayers.indexOf(player.id) === -1) boosterPlayers.push(player.id);
            room.sendAnnouncement("💎 Você comprou o BOOSTER por 75.000 coins! O cargo já está ativo.", player.id, Cor.Roxo, "bold", 2);
        }
    }
    saveProfile(getAuth(player), profile);
    room.sendAnnouncement("✅ Compra realizada: " + item.nome + " | Saldo: " + profile.coins + " coins", player.id, 0x7DFA89, "bold");
    sendGeneralLog("LOJA", player.name + " comprou " + item.nome + " por " + item.preco + " coins.");
    return true;
}

function gerarSenhaAleatoria(prefixo) {
    const caracteres = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    let senha = prefixo;
    for (let i = 0; i < 4; i++) {
        senha += caracteres.charAt(Math.floor(Math.random() * caracteres.length));
    }
    return senha;
}

var animacoesAtivas = {};

function iniciarAvatarAnimado(player, animKey) {
    const anim = animatedAvatars[animKey];
    if (!anim || !player) return false;

    // Cancela qualquer animação anterior do mesmo jogador.
    if (animacoesAtivas[player.id]) {
        clearInterval(animacoesAtivas[player.id].interval);
        delete animacoesAtivas[player.id];
    }

    let frameIndex = 0;

    // Mostra o primeiro frame imediatamente.
    room.setPlayerAvatar(player.id, anim.frames[frameIndex]);
    frameIndex = (frameIndex + 1) % anim.frames.length;

    const intervalId = setInterval(function() {
        const p = room.getPlayer(player.id);

        if (!p) {
            clearInterval(intervalId);
            delete animacoesAtivas[player.id];
            return;
        }

        // Se outra animação substituiu esta, encerra este timer.
        if (!animacoesAtivas[player.id] ||
            animacoesAtivas[player.id].interval !== intervalId) {
            clearInterval(intervalId);
            return;
        }

        room.setPlayerAvatar(player.id, anim.frames[frameIndex]);
        frameIndex = (frameIndex + 1) % anim.frames.length;
    }, anim.interval);

    animacoesAtivas[player.id] = {
        interval: intervalId,
        animacao: animKey,
        frames: anim.frames
    };

    return true;
}

function pararAvatarAnimado(player) {
    if (!player) return false;

    if (animacoesAtivas[player.id]) {
        clearInterval(animacoesAtivas[player.id].interval);
        delete animacoesAtivas[player.id];
        room.setPlayerAvatar(player.id, null);
        return true;
    }

    return false;
}

    var partes = [];

function checkTime() {
    const scores = room.getScores();
    game.scores = scores;
    if (Math.abs(scores.time - scores.timeLimit) <= 0.01 && scores.timeLimit != 0) {
        if (scores.red != scores.blue) {
            if (checkTimeVariable == false) {
                checkTimeVariable = true;
                setTimeout(() => { checkTimeVariable = false; }, 3000);
                scores.red > scores.blue ? endGame(Team.RED) : endGame(Team.BLUE);
                setTimeout(() => { room.stopGame(); }, 2000);
            }
            return;
        }
        goldenGoal = true;
        room.sendAnnouncement(centerText("PRORROGAÇÃO"), null, Cor.Amarelo, "bold");
        room.sendAnnouncement(centerText("Vou dar " + drawTimeLimit * 60 + " segundos!"), null, Cor.Branco, "normal");
        room.sendAnnouncement(centerText("⚽ Primeiro gol vence! ⚽"), null, Cor.Branco, "normal");
    }
    if (scores.time > scores.timeLimit + drawTimeLimit * 60 - 15 && scores.time <= scores.timeLimit + drawTimeLimit * 60) {
        if (checkTimeVariable == false && announced == false) {
            checkTimeVariable = true;
            announced = true;
            setTimeout(() => { checkTimeVariable = false; }, 10);
            room.sendAnnouncement(centerText("⌛ 15 segundos para o empate!"), null, Cor.Amarelo, "bold");
        }
    }
    if (scores.time > (scores.timeLimit + drawTimeLimit * 60)) {
        if (checkTimeVariable == false) {
            checkTimeVariable = true;
            setTimeout(() => { checkTimeVariable = false; }, 10);
            room.sendAnnouncement(centerText("🤝 EMPATE! 2 MINUTOS DE PRORROGAÇÃO SEM GOL."), null, Cor.Amarelo, "bold");
            room.sendAnnouncement(centerText("🔴 Os dois times serão enviados para os espectadores."), null, Cor.Branco, "normal");
            room.setPlayerTeam(-1, Team.SPECTATORS);
            endGame(Team.SPECTATORS);
            room.stopGame();
            goldenGoal = false;
        }
    }
}

function endGame(winner) {
    // Evita contar a mesma partida duas vezes nas estatísticas.
    if (endGameVariable) return;
    endGameVariable = true;
    players.length >= 2 * maxTeamSize ? activateChooseMode() : null;
    const scores = room.getScores();
    game.scores = scores;
    Rposs = Rposs / (Rposs + Bposs);
    Bposs = 1 - Rposs;
    lastWinner = winner;
    if (winner == Team.RED) {
        streak++;
        room.sendAnnouncement(centerText("🏆 Time vermelho venceu! | Sequência de Vitória(s): ") + streak + " 🏆", null, 0xFDC43A);
    } else if (winner == Team.BLUE) {
        streak = 1;
        room.sendAnnouncement(centerText("🏆 Time azul venceu! | Sequência de Vitória(s): ") + streak + " 🏆", null, 0xFDC43A);
    } else {
        streak = 0;
        room.sendAnnouncement("💤 Tempo limite alcançado");
    }
    room.sendAnnouncement(centerText("🏆 FIM DE PARTIDA 🏆"), null, Cor.Branco, "bold");
    room.sendAnnouncement(centerText("" + scores.red + " - " + scores.blue), null, Cor.Branco, "normal");
    room.sendAnnouncement(centerText((Rposs * 100).toPrecision(3).toString() + "% | Posse de bola | " + (Bposs * 100).toPrecision(3).toString() + "%"), null, Cor.Branco, "normal");
    scores.red == 0 ? (scores.blue == 0 ? room.sendAnnouncement("🥅 " + GKList[0].name + " é um homem? não, é uma barreira! " + GKList[1].name + " não tomou gol.", null, 0xFDC43A) : room.sendAnnouncement("🥅 é um homem? não, é uma barreira! " + GKList[1].name + " não tomou nenhum gol", null, 0xFDC43A)) : scores.blue == 0 ? room.sendAnnouncement("🥅 é um homem? não, é uma barreira! " + GKList[0].name + " não tomou nenhum gol", null, 0xFDC43A) : null;
    atualizarSequenciasJogadores(winner);

    // Atualiza estatísticas da temporada antes do reset/recompensa.
    var jogadoresTemporada = allReds.concat(allBlues);
    jogadoresTemporada.forEach(function(p) {
        if (!p) return;
        registrarStatTemporada(p, "game", 1);

        if (winner === Team.RED && p.team === Team.RED) registrarStatTemporada(p, "win", 1);
        else if (winner === Team.BLUE && p.team === Team.BLUE) registrarStatTemporada(p, "win", 1);
        else if (winner === Team.RED || winner === Team.BLUE) registrarStatTemporada(p, "loss", 1);
        else registrarStatTemporada(p, "draw", 1);
    });

    (game.goals || []).forEach(function(g) {
        if (g.striker) registrarStatTemporada(g.striker, "goal", 1);
        if (g.assist) registrarStatTemporada(g.assist, "assist", 1);
    });

    var mvp = calcularMVPPartida();
    if (mvp) {
        registrarStatTemporada(mvp.player, "mvp", 1);
        room.sendAnnouncement(centerText("👑 MVP DA PARTIDA: " + mvp.player.name + " | " + mvp.score + " pontos"), null, Cor.Ouro, "bold", 2);
    }
    var sumula = gerarSumulaPartida();
    enviarSumulaDiscord(sumula);
    sendGeneralLog("PARTIDA ENCERRADA", sumula.replace(/\*\*/g, ""));
    updateStats();
    premiarCoinsDaPartida(mvp);

    // Todo uniforme Premium/Rivals usado na partida é descartado.
    // A próxima partida sempre começa com uniformes normais.
    room.setTeamColors(Team.RED, 0, 0xFFFFFF, [0xFF0000]);
    room.setTeamColors(Team.BLUE, 0, 0xFFFFFF, [0x0000FF]);

    verificarFimTemporada();
}

function handleInactivity() {
    // AFK é baseado na POSIÇÃO real do jogador.
    // Assim funciona para jogador normal, Admin, SuperAdmin e Booster,
    // e também conta desde o início da partida, antes do primeiro toque na bola.
    if (!countAFK) return;

    var jogadoresEmCampo = TeamR.concat(TeamB);

    for (var i = 0; i < jogadoresEmCampo.length; i++) {
        var jogador = jogadoresEmCampo[i];
        if (!jogador || !jogador.position) continue;

        var playerId = jogador.id;
        var pos = jogador.position;
        var anterior = afkPositionState[playerId];

        if (!anterior) {
            afkPositionState[playerId] = { x: pos.x, y: pos.y };
            setActivity(jogador, 0);
            continue;
        }

        var dx = pos.x - anterior.x;
        var dy = pos.y - anterior.y;
        var distancia = Math.sqrt(dx * dx + dy * dy);

        if (distancia >= AFK_MOVIMENTO_MINIMO) {
            // Voltou a se mexer.
            setActivity(jogador, 0);
            delete afkWarnings[playerId];
            afkPositionState[playerId] = { x: pos.x, y: pos.y };
        } else {
            // Ficou praticamente no mesmo lugar: +1 tick (60 ticks = ~1 segundo).
            setActivity(jogador, (getActivity(jogador) || 0) + 1);
        }
    }

    var avisoTicks = 60 * AFK_AVISO_SEGUNDOS;
    var kickTicks = 60 * AFK_KICK_SEGUNDOS;

    for (var j = 0; j < extendedP.length; j++) {
        var registroAFK = extendedP[j];
        var playerId2 = registroAFK[eP.ID];
        var jogador2 = room.getPlayer(playerId2);

        if (!jogador2 || jogador2.team === Team.SPECTATORS) {
            delete afkWarnings[playerId2];
            delete afkPositionState[playerId2];
            continue;
        }

        var atividade = Number(registroAFK[eP.ACT]) || 0;

        if (atividade >= avisoTicks && !afkWarnings[playerId2]) {
            afkWarnings[playerId2] = true;
            room.sendAnnouncement(
                "⚠️ @" + jogador2.name +
                ", você está AFK! Se não se mexer nos próximos 5 segundos, será kickado!",
                playerId2,
                0xF4A404,
                "bold",
                2
            );
        }

        if (atividade >= kickTicks) {
            registroAFK[eP.ACT] = 0;
            delete afkWarnings[playerId2];
            delete afkPositionState[playerId2];

            room.kickPlayer(playerId2, "AFK por " + AFK_KICK_SEGUNDOS + " segundos", false);
        }
    }
}

function updateRoleOnPlayerOut() {
    updateTeams();
    if (room.getScores() != null) {
        var scores = room.getScores();
        if (players.length >= 2 * maxTeamSize && scores.time >= (5 / 6) * game.scores.timeLimit && TeamR.length != TeamB.length) {
            if (TeamR.length < TeamB.length) {
                if (scores.blue - scores.red == 2) {
                    endGame(Team.BLUE);
                    room.sendAnnouncement("[Juiz] Ragequit por parte do time vermelho detectado, partida terminada!", null, 0xbfff00, "normal");
                    setTimeout(() => { room.stopGame(); }, 100);
                    return;
                }
            } else {
                if (scores.red - scores.blue == 2) {
                    endGame(Team.RED);
                    room.sendAnnouncement("[Juiz] Ragequit por parte do time azul detectado, partida terminada!", null, 0xbfff00, "normal");
                    setTimeout(() => { room.stopGame(); }, 100);
                    return;
                }
            }
        }
    }
    if (inChooseMode) {
        if (players.length < 2 * maxTeamSize) {
            loadMap(practiceMap, scoreLimitPractice, timeLimitPractice);
        }
        if (TeamR.length == 0 || TeamB.length == 0) {
            TeamR.length == 0 ? room.setPlayerTeam(teamS[0].id, Team.RED) : room.setPlayerTeam(teamS[0].id, Team.BLUE);
            return;
        }
        if (Math.abs(TeamR.length - TeamB.length) == teamS.length) {
            room.sendAnnouncement("Sem jogadores suficientes, escolha automaticaticamente!!", null, 0xFF0000, "normal");
            deactivateChooseMode();
            resumeGame();
            var b = teamS.length;
            if (TeamR.length > TeamB.length) {
                for (var i = 0; i < b; i++) {
                    setTimeout(() => { room.setPlayerTeam(teamS[0].id, Team.BLUE); }, 5 * i);
                }
            } else {
                for (var i = 0; i < b; i++) {
                    setTimeout(() => { room.setPlayerTeam(teamS[0].id, Team.RED); }, 5 * i);
                }
            }
            return;
        }
        if (streak == 0 && room.getScores() == null) {
            if (Math.abs(TeamR.length - TeamB.length) == 2) {
                room.sendAnnouncement("Equilibrando times...", null, 0xc0bdb1, "normal");
                TeamR.length > TeamB.length ? room.setPlayerTeam(TeamR[TeamR.length - 1].id, Team.SPECTATORS) : room.setPlayerTeam(TeamB[TeamB.length - 1].id, Team.SPECTATORS);
            }
        }
        if (TeamR.length == TeamB.length && teamS.length < 2) {
            deactivateChooseMode();
            resumeGame();
            return;
        }
        capLeft ? choosePlayer() : getSpecList(TeamR.length <= TeamB.length ? TeamR[0] : TeamB[0]);
    }
    balanceTeams();
}

function balanceTeams() {
    if (!inChooseMode) {
        if (players.length == 1 && TeamR.length == 0) {
            quickRestart();
            loadMap(practiceMap, scoreLimitPractice, timeLimitPractice);
            room.setPlayerTeam(players[0].id, Team.RED);
        } else if (Math.abs(TeamR.length - TeamB.length) == teamS.length && teamS.length > 0) {
            const n = Math.abs(TeamR.length - TeamB.length);
            if (TeamR.length > TeamB.length) {
                for (var i = 0; i < n; i++) {
                    room.setPlayerTeam(teamS[i].id, Team.BLUE);
                }
            } else {
                for (var i = 0; i < n; i++) {
                    room.setPlayerTeam(teamS[i].id, Team.RED);
                }
            }
        } else if (Math.abs(TeamR.length - TeamB.length) > teamS.length) {
            const n = Math.abs(TeamR.length - TeamB.length);
            if (players.length == 1) {
                quickRestart();
                loadMap(practiceMap, scoreLimitPractice, timeLimitPractice);
                room.setPlayerTeam(players[0].id, Team.RED);
                return;
            } else if (players.length == 2 * maxTeamSize) {
                quickRestart();
                loadMap(practiceMap, scoreLimitPractice, timeLimitPractice);
            }
            if (players.length == maxTeamSize * 2 - 1) {
                allReds = [];
                allBlues = [];
            }
            if (TeamR.length > TeamB.length) {
                for (var i = 0; i < n; i++) {
                    room.setPlayerTeam(TeamR[TeamR.length - 1 - i].id, Team.SPECTATORS);
                }
            } else {
                for (var i = 0; i < n; i++) {
                    room.setPlayerTeam(TeamB[TeamB.length - 1 - i].id, Team.SPECTATORS);
                }
            }
        } else if (Math.abs(TeamR.length - TeamB.length) < teamS.length && TeamR.length != TeamB.length) {
            room.pauseGame(true);
            activateChooseMode();
            choosePlayer();
        } else if (teamS.length >= 2 && TeamR.length == TeamB.length && TeamR.length < maxTeamSize) {
            if (TeamR.length == 2) {
                quickRestart();
                if (!teamS.length == 2) {
                    loadMap(practiceMap, scoreLimitPractice, timeLimitPractice);
                }
            }
            topBtn();
        }
    }
}

function choosePlayer() {
    clearTimeout(timeOutCap);
    
    // Determina quem é o capitão que vai escolher
    var captain = null;
    var isRedTurn = TeamR.length <= TeamB.length && TeamR.length != 0;
    
    if (isRedTurn) {
        captain = TeamR[0];
    } else if (TeamB.length < TeamR.length && TeamB.length != 0) {
        captain = TeamB[0];
    }

    if (captain) {
        // Mostra a lista numerada no chat para o capitão
        getSpecList(captain);
        
        // Aviso inicial
        room.sendAnnouncement("🎯 " + captain.name + ", digite apenas o NÚMERO do jogador que deseja escolher.", captain.id, 0x00FF00, "bold");

        // Timer de 8 segundos para aviso
        timeOutCap = setTimeout(function(player) {
            if (!inChooseMode) return;
            // Verifica se o jogador ainda é capitão e se ninguém foi escolhido ainda (lógica simplificada)
            room.sendAnnouncement("⚠️ Seja rápido, @" + player.name + "! Se não escolher em 10s, você será kickado!", player.id, 0xFF0000, "bold");
            
            // Timer final de kick (10s após o aviso)
            timeOutCap = setTimeout(function(p) {
                if (inChooseMode && ((isRedTurn && TeamR[0] && TeamR[0].id === p.id) || (!isRedTurn && TeamB[0] && TeamB[0].id === p.id))) {
                     room.kickPlayer(p.id, "Demorou demais para escolher", false);
                }
            }, 10000, player);
        }, 8000, captain);
    }
}

}

function updateStats() {
    // ================= STATS 4x4 =================
    // O fim da partida já é decidido por endGame(). Portanto não precisamos
    // testar scoreLimit/timeLimit novamente aqui.
    // Usamos o snapshot matchReds/matchBlues para nunca perder os participantes
    // originais do 4x4 por causa de substituição, saída ou mudança de time.

    if (!game) return;
    if (maxTeamSize !== 4) return;

    var reds = matchReds.length === maxTeamSize ? matchReds : allReds.slice(0, maxTeamSize);
    var blues = matchBlues.length === maxTeamSize ? matchBlues : allBlues.slice(0, maxTeamSize);

    if (reds.length !== maxTeamSize || blues.length !== maxTeamSize) {
        console.log("[STATS 4x4] Partida ignorada: snapshot incompleto.",
            "RED:", reds.length, "BLUE:", blues.length);
        return;
    }

    var todos = reds.concat(blues);
    var seen = {};

    // Partida + resultado.
    todos.forEach(function(p) {
        if (!p || seen[p.id]) return;
        seen[p.id] = true;

        var auth = p.auth || getAuth(p);
        if (!auth) {
            console.log("[STATS 4x4] Jogador sem AUTH, ignorado:", p.name);
            return;
        }
        var raw = localStorage.getItem(auth);
        var stats = raw ? JSON.parse(raw) :
            [0, 0, 0, 0, "0.00", 0, 0, 0, 0, "0.00", "player", p.name];

        while (stats.length <= Ss.NK) stats.push(0);

        stats[Ss.GA] = Number(stats[Ss.GA]) || 0;
        stats[Ss.WI] = Number(stats[Ss.WI]) || 0;
        stats[Ss.DR] = Number(stats[Ss.DR]) || 0;
        stats[Ss.LS] = Number(stats[Ss.LS]) || 0;
        stats[Ss.GL] = Number(stats[Ss.GL]) || 0;
        stats[Ss.AS] = Number(stats[Ss.AS]) || 0;
        stats[Ss.GK] = Number(stats[Ss.GK]) || 0;
        stats[Ss.CS] = Number(stats[Ss.CS]) || 0;
        stats[Ss.NK] = p.name;

        stats[Ss.GA]++;

        if (lastWinner === Team.RED) {
            if (reds.some(function(x) { return x && x.id === p.id; })) stats[Ss.WI]++;
            else stats[Ss.LS]++;
        } else if (lastWinner === Team.BLUE) {
            if (blues.some(function(x) { return x && x.id === p.id; })) stats[Ss.WI]++;
            else stats[Ss.LS]++;
        } else {
            stats[Ss.DR]++;
        }

        var jogosDecisivos = stats[Ss.WI] + stats[Ss.LS] + stats[Ss.DR];
        stats[Ss.WR] = jogosDecisivos > 0
            ? (100 * stats[Ss.WI] / jogosDecisivos).toFixed(2)
            : "0.00";

        localStorage.setItem(auth, JSON.stringify(stats));
    });

    // Gols e assistências.
    (game.goals || []).forEach(function(goal) {
        if (goal.striker) {
            var authGol = goal.striker.auth || getAuth(goal.striker);
            if (!authGol) {
                console.log("[STATS 4x4] Gol sem AUTH:", goal.striker.name);
            } else {
            var rawGol = localStorage.getItem(authGol);
            if (!rawGol) {
                localStorage.setItem(authGol, JSON.stringify(
                    [0,0,0,0,"0.00",0,0,0,0,"0.00","player",goal.striker.name]
                ));
                rawGol = localStorage.getItem(authGol);
            }
            var stGol = JSON.parse(rawGol);
                stGol[Ss.GL] = (Number(stGol[Ss.GL]) || 0) + 1;
                localStorage.setItem(authGol, JSON.stringify(stGol));
            }
        }

        if (goal.assist) {
            var authAssist = goal.assist.auth || getAuth(goal.assist);
            if (!authAssist) {
                console.log("[STATS 4x4] Assistência sem AUTH:", goal.assist.name);
            } else {
            var rawAssist = localStorage.getItem(authAssist);
            if (!rawAssist) {
                localStorage.setItem(authAssist, JSON.stringify(
                    [0,0,0,0,"0.00",0,0,0,0,"0.00","player",goal.assist.name]
                ));
                rawAssist = localStorage.getItem(authAssist);
            }
            var stAssist = JSON.parse(rawAssist);
                stAssist[Ss.AS] = (Number(stAssist[Ss.AS]) || 0) + 1;
                localStorage.setItem(authAssist, JSON.stringify(stAssist));
            }
        }
    });

    // GK / Clean Sheet.
    if (GKList && GKList.length) {
        GKList.forEach(function(gk) {
            if (!gk || !gk.auth) return;

            var pertence = todos.some(function(p) { return p && p.id === gk.id; });
            if (!pertence) return;

            var authGK = getAuth(gk);
            var rawGK = localStorage.getItem(authGK);
            if (!rawGK) {
                localStorage.setItem(authGK, JSON.stringify(
                    [0,0,0,0,"0.00",0,0,0,0,"0.00","player",gk.name]
                ));
                rawGK = localStorage.getItem(authGK);
            }

            var stGK = JSON.parse(rawGK);
            stGK[Ss.GK] = (Number(stGK[Ss.GK]) || 0) + 1;

            var sofreu = (gk.team === Team.RED)
                ? Number(game.scores.blue) || 0
                : Number(game.scores.red) || 0;

            if (sofreu === 0) stGK[Ss.CS] = (Number(stGK[Ss.CS]) || 0) + 1;

            stGK[Ss.CP] = stGK[Ss.GK] > 0
                ? (100 * stGK[Ss.CS] / stGK[Ss.GK]).toFixed(2)
                : "0.00";

            localStorage.setItem(authGK, JSON.stringify(stGK));
        });
    }

    console.log("[STATS 4x4] OK:",
        todos.length + " jogadores",
        "| gols:", (game.goals || []).length,
        "| vencedor:", lastWinner,
        "| stats salvos por AUTH.");

    sendGeneralLog(
        "STATS 4x4",
        todos.length + " jogadores atualizados | " +
        (game.goals || []).length + " gols/assistências processados."
    );
}

function findGK() {
    var tab = [[-1, ""], [-1, ""]];
    for (var i = 0; i < extendedP.length; i++) {
        if (room.getPlayer(extendedP[i][eP.ID]) != null && room.getPlayer(extendedP[i][eP.ID]).team == Team.RED) {
            if (tab[0][0] < extendedP[i][eP.GK]) {
                tab[0][0] = extendedP[i][eP.GK];
                tab[0][1] = room.getPlayer(extendedP[i][eP.ID]);
            }
        } else if (room.getPlayer(extendedP[i][eP.ID]) != null && room.getPlayer(extendedP[i][eP.ID]).team == Team.BLUE) {
            if (tab[1][0] < extendedP[i][eP.GK]) {
                tab[1][0] = extendedP[i][eP.GK];
                tab[1][1] = room.getPlayer(extendedP[i][eP.ID]);
            }
        }
    }
    GKList = [tab[0][1], tab[1][1]];
}

room.onPlayerTeamChange = function(changedPlayer, byPlayer) {
    if (changedPlayer.id == 0) {
        room.setPlayerTeam(0, Team.SPECTATORS);
        return;
    }
    if (getAFK(changedPlayer) && changedPlayer.team != Team.SPECTATORS) {
        room.setPlayerTeam(changedPlayer.id, Team.SPECTATORS);
        room.sendChat(changedPlayer.name + " está AFK!");
        return;
    }
    updateTeams();
    if (room.getScores() != null) {
        var scores = room.getScores();
        if (changedPlayer.team != Team.SPECTATORS && scores.time <= (3 / 4) * scores.timeLimit && Math.abs(scores.blue - scores.red) < 2) {
            if (changedPlayer.team == Team.RED) {
                if (!allReds.some(function(p) { return p && p.id === changedPlayer.id; })) allReds.push(changedPlayer);
            } else {
                if (!allBlues.some(function(p) { return p && p.id === changedPlayer.id; })) allBlues.push(changedPlayer);
            }
        }
    }
    if (changedPlayer.team == Team.SPECTATORS) {
        setActivity(changedPlayer, 0);
        delete afkWarnings[changedPlayer.id];
    }
    if (inChooseMode && resettingTeams == false && byPlayer.id == 0) {
        if (Math.abs(TeamR.length - TeamB.length) == teamS.length) {
            deactivateChooseMode();
            resumeGame();
            var b = teamS.length;
            if (TeamR.length > TeamB.length) {
                for (var i = 0; i < b; i++) {
                    setTimeout(() => { room.setPlayerTeam(teamS[0].id, Team.BLUE); }, 200 * i);
                }
            } else {
                for (var i = 0; i < b; i++) {
                    setTimeout(() => { room.setPlayerTeam(teamS[0].id, Team.RED); }, 200 * i);
                }
            }
            return;
        } else if ((TeamR.length == maxTeamSize && TeamB.length == maxTeamSize) || (TeamR.length == TeamB.length && teamS.length < 2)) {
            deactivateChooseMode();
            resumeGame();
        } else if (TeamR.length <= TeamB.length && redCaptainChoice != "") {
            redCaptainChoice == "top" ? room.setPlayerTeam(teamS[0].id, Team.RED) : redCaptainChoice == "random" ? room.setPlayerTeam(teamS[getRandomInt(teamS.length)].id, Team.RED) : room.setPlayerTeam(teamS[teamS.length - 1].id, Team.RED);
            return;
        } else if (TeamB.length < TeamR.length && blueCaptainChoice != "") {
            blueCaptainChoice == "top" ? room.setPlayerTeam(teamS[0].id, Team.BLUE) : blueCaptainChoice == "random" ? room.setPlayerTeam(teamS[getRandomInt(teamS.length)].id, Team.BLUE) : room.setPlayerTeam(teamS[teamS.length - 1].id, Team.BLUE);
            return;
        } else {
            choosePlayer();
        }
    }
}

room.onPlayerLeave = function(player) {
    delete floodControle[player.id];

    sendEventLog(player.name, "LEAVE", "ID: " + player.id);
    sendGeneralLog("LEAVE", player.name + " saiu da sala.");
    if (TeamR.findIndex((red) => red.id == player.id) == 0 && inChooseMode && TeamR.length <= TeamB.length) {
        choosePlayer();
        capLeft = true;
        setTimeout(() => { capLeft = false; }, 10);
    }
    if (TeamB.findIndex((blue) => blue.id == player.id) == 0 && inChooseMode && TeamB.length < TeamR.length) {
        choosePlayer();
        capLeft = true;
        setTimeout(() => { capLeft = false; }, 10);
    }
    let superIndex = superAdmins.indexOf(player.id);
    if (superIndex > -1) superAdmins.splice(superIndex, 1);
    let boosterIndex = boosterPlayers.indexOf(player.id);
    if (boosterIndex > -1) boosterPlayers.splice(boosterIndex, 1);
    setActivity(player, 0);
    delete afkWarnings[player.id];
    delete afkPositionState[player.id];
    if (animacoesAtivas[player.id]) {
        clearInterval(animacoesAtivas[player.id].interval);
        delete animacoesAtivas[player.id];
    }
    updateRoleOnPlayerOut();
}

room.onPlayerChat = function(player, message) {
    sendChatLog(player.name, message, player.admin === true, boosterPlayers.indexOf(player.id) !== -1);

    // TEAM CHAT
    if (message.length > 1 && message[0].toLowerCase() == 't' && message[1] == ' ') {
        if (player.team != 0) {
            room.getPlayerList().forEach((element) => {
                if (element.team == player.team) room.sendAnnouncement("[TEAM CHAT] " + player.name + ": " + message.substr(2), element.id, (player.team == 1 ? 16725591 : 3261685), "bold", 0);
            });
            return false;
        } else {
            room.sendAnnouncement("Você não está em um time.", player.id);
        }
    }

    // =====================================================
    // ★★★ ANTI-FLOOD INTELIGENTE ★★★
    // Não pune uma única mensagem. Só atua quando o jogador
    // envia várias mensagens em poucos segundos.
    // =====================================================
    var agoraFlood = Date.now();

    if (!floodControle[player.id]) {
        floodControle[player.id] = [];
    }

    // Mantém somente mensagens enviadas dentro da janela configurada.
    floodControle[player.id] = floodControle[player.id].filter(function(t) {
        return (agoraFlood - t) <= FLOOD_JANELA_MS;
    });

    floodControle[player.id].push(agoraFlood);

    var qtdFlood = floodControle[player.id].length;

    // Aviso na 4ª mensagem rápida.
    if (qtdFlood === FLOOD_AVISO_QTD && !getMute(player)) {
        room.sendAnnouncement(
            "⚠️ " + player.name + ", diminua a velocidade do chat. Mais uma mensagem rápida poderá causar mute!",
            player.id, 0xFFC12F, "bold", 1
        );
    }

    // 5 mensagens dentro da janela = mute automático de 5 minutos.
    // Admin também pode ser mutado pelo anti-flood; o Admin continua
    // podendo usar !unmute para retirar o mute antes do tempo.
    if (qtdFlood >= FLOOD_MUTE_QTD && !getMute(player)) {
        setMute(player, true);

        sendModerationLog(
            "AUTO_MUTE",
            player.name,
            "Sistema",
            "Flood: " + qtdFlood + " mensagens em " + (FLOOD_JANELA_MS / 1000) + "s. Mute de 5 minutos."
        );

        room.sendAnnouncement(
            "🔇 " + player.name + " foi mutado automaticamente por flood durante 5 minutos!",
            null, 0x9B59B6, "bold", 2
        );

        setTimeout(function(targetId) {
            var p = room.getPlayer(targetId);

            if (p && getMute(p)) {
                setMute(p, false);

                room.sendAnnouncement(
                    "🔊 " + p.name + " foi desmutado automaticamente após 5 minutos.",
                    null, 0x7DFA89, "bold", 1
                );

                sendModerationLog(
                    "UNMUTE_AUTO",
                    p.name,
                    "Sistema",
                    "Os 5 minutos do anti-flood terminaram."
                );
            }
        }, FLOOD_MUTE_MS, player.id);

        // Limpa o contador depois do mute.
        floodControle[player.id] = [];
    }
    // =====================================================
    // ★★★ FIM DO ANTI-FLOOD ★★★
    // =====================================================

    // SELEÇÕES / UNIFORMES DE CAPITÃO

        room.sendAnnouncement("🎮 Partidas: " + (Number(stats[Ss.GA]) || 0), player.id, 0xFFFFFF, "normal");
        room.sendAnnouncement("⚽ Gols: " + (Number(stats[Ss.GL]) || 0) + " | 👟 Assistências: " + (Number(stats[Ss.AS]) || 0), player.id, 0xFFFFFF, "normal");
        room.sendAnnouncement("🧤 Partidas como GK: " + (Number(stats[Ss.GK]) || 0), player.id, 0xFFFFFF, "normal");
        room.sendAnnouncement("📈 WR: " + (stats[Ss.WR] || "0.00") + "%", player.id, 0xFFFFFF, "normal");
        room.sendAnnouncement("「👓」Essas estatísticas contam somente partidas oficiais 4x4.", player.id, 0x7DFA89, "bold");
        return false;
    }

    if (["!mestats"].includes(message[0].toLowerCase())) {
        var stats;
        localStorage.getItem(getAuth(player)) ? stats = JSON.parse(localStorage.getItem(getAuth(player))) : stats = [0, 0, 0, 0, "0.00", 0, 0, 0, 0, "0.00"];

        room.sendAnnouncement("⸢📄⸥ O Jogador " + player.name + " mostrou seus stats!", null, 0xFF7900, "bold");
        room.sendAnnouncement("🎮 Partidas: " + (Number(stats[Ss.GA]) || 0), null, 0xFFFFFF, "normal");
        room.sendAnnouncement("⚽ Gols: " + (Number(stats[Ss.GL]) || 0) + " | 👟 Assistências: " + (Number(stats[Ss.AS]) || 0), null, 0xFFFFFF, "normal");
        room.sendAnnouncement("🧤 Partidas como GK: " + (Number(stats[Ss.GK]) || 0), null, 0xFFFFFF, "normal");
        room.sendAnnouncement("📈 WR: " + (stats[Ss.WR] || "0.00") + "%", null, 0xFFFFFF, "normal");
        return false;
    }

    // ★★★ COMANDO !MUTELIST / !MUTES ★★★
    if (["!mutes", "!mutelist"].includes(message[0].toLowerCase())) {
        var cstm = " Lista de mutados: ";
        var temMutado = false;
        for (var i = 0; i < extendedP.length; i++) {
            if (room.getPlayer(extendedP[i][eP.ID]) != null && getMute(room.getPlayer(extendedP[i][eP.ID]))) {
                temMutado = true;
                if (140 - cstm.length < (room.getPlayer(extendedP[i][eP.ID]).name + "[" + (extendedP[i][eP.ID]) + "], ").length) {
                    room.sendChat(cstm, player.id);
                    cstm = "... ";
                }
                cstm += room.getPlayer(extendedP[i][eP.ID]).name + "[" + (extendedP[i][eP.ID]) + "], ";
            }
        }
        if (!temMutado) {
            room.sendChat(" Não há ninguém na lista de mutados!", player.id);
            return false;
        }
        cstm = cstm.substring(0, cstm.length - 2);
        cstm += ".";
        room.sendChat(cstm, player.id);
        return false;
    }

    // ★★★ COMANDO !MUTE (ADMIN) ★★★
    // Uso:
    // !mute #1 5   -> muta o ID 1 por 5 minutos
    // !mute #1     -> muta o ID 1 por 5 minutos
    // !mute 1 5   -> também aceita sem #
    if (["!mute"].includes(message[0].toLowerCase())) {
        if (!player.admin) return false;

        var alvoMuteRaw = message[1];
        var minutosMute = Number(message[2]);

        if (!alvoMuteRaw) {
            room.sendAnnouncement("❌ Use: !mute #ID minutos | Ex.: !mute #1 5", player.id, 0xFF6B6B, "bold");
            return false;
        }

        var alvoIdTexto = String(alvoMuteRaw).replace(/^#/, "");
        var alvoId = Number(alvoIdTexto);

        if (!Number.isInteger(alvoId) || alvoId <= 0 || !room.getPlayer(alvoId)) {
            room.sendAnnouncement("❌ Jogador não encontrado. Use o ID, por exemplo #1.", player.id, 0xFF6B6B, "bold");
            return false;
        }

        var alvoMute = room.getPlayer(alvoId);

        if (alvoMute.admin) {
            room.sendAnnouncement("❌ Você não pode mutar outro Admin.", player.id, 0xFF6B6B, "bold");
            return false;
        }

        if (getMute(alvoMute)) {
            room.sendAnnouncement("⚠️ " + alvoMute.name + " já está mutado.", player.id, 0xFFC12F, "bold");
            return false;
        }

        if (!Number.isFinite(minutosMute) || minutosMute <= 0) minutosMute = 5;
        minutosMute = Math.floor(minutosMute);

        setMute(alvoMute, true);
        sendModerationLog("MUTE", alvoMute.name, player.name, "Comando administrativo: " + minutosMute + " minuto(s).");

        room.sendAnnouncement(
            "🔇 " + alvoMute.name + " foi mutado por " + minutosMute + " minuto(s).",
            null, 0x9B59B6, "bold", 2
        );

        setTimeout(function(targetId, targetName) {
            var p = room.getPlayer(targetId);

            // Só remove o mute automático se ele ainda estiver mutado.
            if (p && getMute(p)) {
                setMute(p, false);
                room.sendAnnouncement(
                    "🔊 " + targetName + " foi desmutado automaticamente.",
                    null, 0x7DFA89, "bold", 1
                );
                sendModerationLog("UNMUTE_AUTO", targetName, "Sistema", "Tempo do mute encerrado.");
            }
        }, minutosMute * 60 * 1000, alvoMute.id, alvoMute.name);

        return false;
    }

    // ★★★ COMANDO !UNMUTE (ADMIN) ★★★
    // Uso:
    // !unmute #1
    // !unmute 1
    // !unmute all
    if (["!unmute"].includes(message[0].toLowerCase())) {
        if (!player.admin) return false;

        var alvoUnmuteRaw = message[1];

        if (!alvoUnmuteRaw) {
            room.sendAnnouncement("❌ Use: !unmute #ID | !unmute all", player.id, 0xFF6B6B, "bold");
            return false;
        }

        if (String(alvoUnmuteRaw).toLowerCase() === "all") {
            extendedP.forEach(function(ePlayer) {
                ePlayer[eP.MUTE] = false;
            });

            room.sendAnnouncement("🔊 Todos os jogadores foram desmutados.", null, 0x7DFA89, "bold", 1);
            sendModerationLog("UNMUTE_ALL", "Todos", player.name, "Todos os mutes removidos.");
            return false;
        }

        var alvoUnmuteId = Number(String(alvoUnmuteRaw).replace(/^#/, ""));

        if (!Number.isInteger(alvoUnmuteId) || alvoUnmuteId <= 0 || !room.getPlayer(alvoUnmuteId)) {
            room.sendAnnouncement("❌ Jogador não encontrado. Use o ID, por exemplo #1.", player.id, 0xFF6B6B, "bold");
            return false;
        }

        var alvoUnmute = room.getPlayer(alvoUnmuteId);

        if (!getMute(alvoUnmute)) {
            room.sendAnnouncement("⚠️ " + alvoUnmute.name + " não está mutado.", player.id, 0xFFC12F, "bold");
            return false;
        }

        setMute(alvoUnmute, false);
        room.sendAnnouncement("🔊 " + alvoUnmute.name + " foi desmutado por " + player.name + ".", null, 0x7DFA89, "bold", 1);
        sendModerationLog("UNMUTE", alvoUnmute.name, player.name, "Comando administrativo.");

        return false;
    }

    if (["!banlist", "!bans"].includes(message[0].toLowerCase())) {
        if (banList.length == 0) {
            room.sendChat(" Não há ninguém na lista de banidos!", player.id);
            return false;
        }
        var cstm = " Lista de banidos: ";
        for (var i = 0; i < banList.length; i++) {
            if (140 - cstm.length < (banList[i][0] + "[" + (banList[i][1]) + "], ").length) {
                room.sendChat(cstm, player.id);
                cstm = "... ";
            }
            cstm += banList[i][0] + "[" + (banList[i][1]) + "], ";
        }
        cstm = cstm.substring(0, cstm.length - 2);
        cstm += ".";
        room.sendChat(cstm, player.id);
        return false;
    }

    if (["!clearbans"].includes(message[0].toLowerCase())) {
        if (player.admin) {
            if (message.length == 1) {
                room.clearBans();
                room.sendChat("Bans removidos!");
                banList = [];
                sendModerationLog("CLEAR_BANS", "Lista de bans", player.name, "Todos os bans removidos");
            }
            if (message.length == 2) {
                if (!Number.isNaN(Number.parseInt(message[1]))) {
                    if (Number.parseInt(message[1]) > 0) {
                        ID = Number.parseInt(message[1]);
                        room.clearBan(ID);
                        setTimeout(() => {
                            banList = banList.filter((array) => array[1] != ID);
                        }, 20);
                    }
                }
            }
        }
        return false;
    }

    if (["!bb", "!bye", "!cya", "!gn"].includes(message[0].toLowerCase())) {
        room.kickPlayer(player.id, "👋 Até mais!", false);
        return false;
    }

    // BAN AUTOMÁTICO: racismo e uso de "PCD" como ofensa
    var mensagemModeracao = String(originalMessage || "").toLowerCase()
        .normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    // Detecta SOMENTE termos de racismo/PCD como palavras ou expressões inteiras.
    // Xingamentos comuns (ex.: lixo, merda, porra, cu, etc.) NÃO geram kick/ban.
    var racismoDetectado = xingo.some(function(term) {
        var t = String(term).toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim();
        if (!t) return false;
        var escaped = t.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        return new RegExp("(^|\\s)" + escaped + "($|\\s|[!?.,;:])", "i").test(mensagemModeracao);
    });
    var pcdDetectado = pcdTermos.some(function(term) {
        var t = String(term).toLowerCase().replace(/\./g, "");
        return new RegExp("(^|\\s)" + t + "($|\\s|[!?.,;:])", "i").test(mensagemModeracao.replace(/\./g, ""));
    });
    if (racismoDetectado || pcdDetectado) {
        var motivoBan = racismoDetectado ? "❌ Racismo não é tolerado nesta sala." : "❌ Ofensas usando PCD não são toleradas nesta sala.";
        room.kickPlayer(player.id, motivoBan, true);
        sendEventLog(player.name, "AUTO_BAN", motivoBan);
        return false;
    }
    if (link.includes(message[0])) {
        room.sendAnnouncement("Você não pode enviar links aqui, " + player.name, player.id, Cor.Warn, "italic", 2);
        return false;
    }

    if (message[0][0] == "!") {
        return false;
    }

    // ★★★ PICK 4x4: ESCOLHA EXCLUSIVAMENTE POR NÚMERO ★★★
    if (TeamR.length != 0 && TeamB.length != 0 && inChooseMode) {
        if (player.id == TeamR[0].id || player.id == TeamB[0].id) {
            // Verifica se é um número
            if (!Number.isNaN(Number.parseInt(message[0]))) {
                var num = Number.parseInt(message[0]);
                
                if (TeamR.length <= TeamB.length && player.id == TeamR[0].id) {
                     if (num > teamS.length || num < 1) {
                        room.sendAnnouncement("[⚠️] Ops! O número que você escolheu é inválido.", player.id, 0xfaca29, "normal");
                        return false;
                    } else {
                        room.setPlayerTeam(teamS[num - 1].id, Team.RED);
                        room.sendAnnouncement(player.name + " escolheu o(a): " + teamS[num - 1].name + "!", null, 0x55bae2, "normal");
                        clearTimeout(timeOutCap); // Reseta o timer ao escolher
                        choosePlayer(); // Chama o próximo
                        return false;
                    }
                }
                
                if (TeamR.length > TeamB.length && player.id == TeamB[0].id) {
                    if (num > teamS.length || num < 1) {
                        room.sendAnnouncement("[⚠️] Ops! O número que você escolheu é inválido.", player.id, 0xfaca29, "normal");
                        return false;
                    } else {
                        room.setPlayerTeam(teamS[num - 1].id, Team.BLUE);
                        room.sendAnnouncement(player.name + " escolheu o(a): " + teamS[num - 1].name + "!", null, 0x55bae2, "normal");
                        clearTimeout(timeOutCap); // Reseta o timer ao escolher
                        choosePlayer(); // Chama o próximo
                        return false;
                    }
                }
            }
            
            // Nenhum comando textual é aceito durante o Pick 4x4.
        }
    }

    // Verificar se está mutado.
    // O jogador fica em silêncio enquanto durar o mute.
    // Não envia aviso a cada mensagem para evitar spam visual.
    if (getMute(player)) {
        return false;
    }

    // Slow mode
    if (slowMode > 0) {
        if (!player.admin) {
            if (!SMSet.has(player.id)) {
                SMSet.add(player.id);
                setTimeout((number) => {
                    SMSet.delete(number);
                }, slowMode * 1000, player.id);
            } else {
                return false;
            }
        }
    }

    // ================= CHAT FINAL =================
    // Primeiro verifica o CARGO pelo AUTH. Não depende de localStorage/stats.
    // Assim SuperAdmin/Admin/Booster sempre usam o emoji + cor escolhidos.
    // Jogador normal sempre fica branco.
    var cargoChat = cargoInfo(player);
    var rankEmojiChat = "📄";
    var chatColorChat = 0xFFFFFF;

    if (cargoChat && (cargoChat.nome === "SuperAdmin" ||
                      cargoChat.nome === "Admin" ||
                      cargoChat.nome === "Booster")) {
        rankEmojiChat = cargoChat.emoji ||
            (cargoChat.nome === "SuperAdmin" ? "👑" :
             cargoChat.nome === "Admin" ? "🛡️" : "💎");
        chatColorChat = cargoChat.cor;
        if (typeof chatColorChat !== "number") {
            chatColorChat = hexParaCor(chatColorChat, 0xFFFFFF);
        }
    }

    // Mensagem inteira recebe a mesma cor do cargo.
    room.sendAnnouncement(
        "⸢" + rankEmojiChat + "⸥ " + player.name + ": " + originalMessage,
        null,
        chatColorChat,
        "normal"
    );
    return false;
}

room.onGameStart = function(byPlayer) {
    if (!recordingActive && sendRecWebhookURL) {
        try { room.startRecording(); recordingActive = true; } catch (e) { console.log("[REC] Erro ao iniciar gravação:", e); }
    }
    sendEventLog("SALA", "GAME_START", byPlayer ? ("por " + byPlayer.name) : "automático");
    game = new Game(Date.now(), room.getScores(), []);
    coinsAwardedForGame = false;
    countAFK = true;
    activePlay = false;
    goldenGoal = false;
    afkPositionState = {};
    afkWarnings = {};
    allReds.concat(allBlues).forEach(function(p) {
        if (p) setActivity(p, 0);
    });
    endGameVariable = false;
    lastPlayersTouched = [null, null];
    Rposs = 0;
    Bposs = 0;
    GKList = [];
    allReds = [];
    allBlues = [];
    matchReds = [];
    matchBlues = [];

    // Snapshot oficial da partida. No 4x4, estes 8 jogadores são os
    // participantes que receberão as estatísticas, mesmo que alguém saia
    // ou troque de time depois.
    if (TeamR.length === maxTeamSize && TeamB.length === maxTeamSize) {
        matchReds = TeamR.slice(0, maxTeamSize);
        matchBlues = TeamB.slice(0, maxTeamSize);
        allReds = matchReds.slice();
        allBlues = matchBlues.slice();
    }

    room.sendAnnouncement(centerText("🥅 PARTIDA INICIANDO 🥅"), null, Cor.Branco, "bold");
    room.sendAnnouncement(centerText("Quer mudar o uniforme? Digite '!uni'"), null, 0x2EF55D, "bold");
    room.sendAnnouncement(centerText("[💬] Use 't' mensagem para conversar com a sua equipe!"), null, 0x5EE7FF);
    for (var i = 0; i < extendedP.length; i++) {
        extendedP[i][eP.GK] = 0;
        extendedP[i][eP.ACT] = 0;
        room.getPlayer(extendedP[i][eP.ID]) == null ? extendedP.splice(i, 1) : null;
    }
    deactivateChooseMode();
}

// ★★★ SISTEMA WINNER STAYS (VENCEDOR FICA) ★★★
room.onGameStop = function(byPlayer) {
    if ((!byPlayer || byPlayer.id == 0) && endGameVariable) {
        var recData = null;
        if (recordingActive) {
            try { recData = room.stopRecording(); } catch (e) { console.log("[REC] Erro ao parar gravação:", e); }
            recordingActive = false;
        }
        if (recData) enviarGravacaoDiscord(recData, gerarSumulaPartida());
        sendEventLog("SALA", "GAME_STOP", "gravação enviada");
        sendGeneralLog("GAME_STOP", "Partida encerrada. Replay: " + (recData ? "sim" : "não"));
        updateTeams();

        if (inChooseMode) {
            if (players.length == 2 * maxTeamSize) {
                inChooseMode = false;
                resetBtn();
                for (var i = 0; i < maxTeamSize; i++) {
                    setTimeout(() => { randomBtn(); }, 400 * i);
                }
                setTimeout(() => { room.startGame(); }, 2000);
            } else {
                // ★★★ WINNER STAYS: VENCEDOR FICA, PERDEDOR VAI PRO SPEC ★★★
                if (lastWinner == Team.RED) {
                    // Red ganhou: Blue vai pro spec, Red fica
                    blueToSpecBtn();
                } else if (lastWinner == Team.BLUE) {
                    // Blue ganhou: Red vai pro spec, Blue vira Red e fica
                    redToSpecBtn();
                    blueToRedBtn();
                } else {
                    resetBtn();
                }
                setTimeout(() => { topBtn(); }, 500);
            }
        } else {
            if (players.length == 2) {
                if (lastWinner == Team.BLUE) {
                    room.setPlayerTeam(TeamB[0].id, Team.RED);
                    room.setPlayerTeam(TeamR[0].id, Team.BLUE);
                }
                setTimeout(() => { room.startGame(); }, 2000);
            } else if (players.length == 3 || players.length >= 2 * maxTeamSize + 1) {
                if (lastWinner == Team.RED) {
                    blueToSpecBtn();
                } else {
                    redToSpecBtn();
                    blueToRedBtn();
                }
                setTimeout(() => { topBtn(); }, 200);
                setTimeout(() => { room.startGame(); }, 2000);
            } else if (players.length == 4) {
                resetBtn();
                setTimeout(() => {
                    randomBtn();
                    setTimeout(() => { randomBtn(); }, 500);
                }, 500);
                setTimeout(() => { room.startGame(); }, 2000);
            } else if (players.length == 5 || players.length >= 2 * maxTeamSize + 1) {
                if (lastWinner == Team.RED) {
                    blueToSpecBtn();
                } else {
                    redToSpecBtn();
                    blueToRedBtn();
                }
                setTimeout(() => { topBtn(); }, 200);
                activateChooseMode();
            } else if (players.length == 6) {
                resetBtn();
                setTimeout(() => {
                    randomBtn();
                    setTimeout(() => {
                        randomBtn();
                        setTimeout(() => { randomBtn(); }, 500);
                    }, 500);
                }, 500);
                setTimeout(() => { room.startGame(); }, 2000);
            }
        }
    }
}

    activePlay = false;
    countAFK = false;
    const scores = room.getScores();
    game.scores = scores;

    if (lastPlayersTouched[0] != null && lastPlayersTouched[0].team == team) {
        if (lastPlayersTouched[1] != null && lastPlayersTouched[1].team == team) {
            room.sendAnnouncement(centerText("TOCA A MÚÚSICAAA, É GOOOOOL!!!"), null, Cor.Verde, "bold");

            room.sendAnnouncement(centerText("Velocidade do chute: " + ballSpeed.toPrecision(4).toString() + " km/h"), null, Cor.Branco, "normal");
            game.goals.push(new Goal(scores.time, team, lastPlayersTouched[0], null));
        }
        if (goalMaker) {
            setTimeout(function() {
                room.setPlayerAvatar(goalMaker, "🎯");
                setTimeout(function() {
                    room.setPlayerAvatar(goalMaker, "⚽");
                    setTimeout(function() {
                        room.setPlayerAvatar(goalMaker, null);
                    }, 3000);
                }, 1200);
            }, 1);
        }
        if (lastPlayersTouched[1] != null && lastPlayersTouched[1].team == team) {
            let goalAssist = lastPlayersTouched[1].id;
            setTimeout(function() {
                room.setPlayerAvatar(goalAssist, "🤝");
                setTimeout(function() {
                    room.setPlayerAvatar(goalAssist, "👟");
                    setTimeout(function() {
                        room.setPlayerAvatar(goalAssist, null);
                    }, 2500);
                }, 1000);
            }, 1);
        }
    } else {
        room.sendAnnouncement(centerText("🤦‍♂️ É GOOOOOL CONTRA!! 🤦‍♂️"), null, Cor.Amarelo, "bold");
        room.sendAnnouncement(centerText("🤡 Gol de " + (lastPlayersTouched[0] ? lastPlayersTouched[0].name : "alguém") + " 🤡"), null, Cor.Branco, "bold");
        room.sendAnnouncement(centerText("Velocidade do chute: " + ballSpeed.toPrecision(4).toString() + " km/h"), null, Cor.Branco, "normal");
        game.goals.push(new Goal(scores.time, team, null, null));
        if (goalMaker) {
            setTimeout(function() {
                room.setPlayerAvatar(goalMaker, "🤦‍♂️");
                setTimeout(function() {
                    room.setPlayerAvatar(goalMaker, "🤡");
                    setTimeout(function() {
                        room.setPlayerAvatar(goalMaker, null);
                    }, 3000);
                }, 1000);
            }, 1);
            golcontra(lastPlayersTouched[0]);
        }
    }

    if (scores.scoreLimit != 0 && (scores.red == scores.scoreLimit || scores.blue == scores.scoreLimit && scores.blue > 0 || goldenGoal == true)) {
        endGame(team);
        goldenGoal = false;
        setTimeout(() => { room.stopGame(); }, 1000);
    }
}

room.onPlayerAdminChange = function(changedPlayer, byPlayer) {
    if (getMute(changedPlayer) && changedPlayer.admin) {
        room.sendChat(changedPlayer.name + " foi desmutado.");
        setMute(changedPlayer, false);
    }
    if (byPlayer.id != 0 && localStorage.getItem(getAuth(byPlayer)) && JSON.parse(localStorage.getItem(getAuth(byPlayer)))[Ss.RL] == "admin") {
        room.sendChat("Você não tem permissão para nomear um jogador como administrador!", byPlayer.id);
        room.setPlayerAdmin(changedPlayer.id, false);
    }
}
